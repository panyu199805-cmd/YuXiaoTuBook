```python
from fastapi import Response
from typing import Any, Optional

class Res:
    @staticmethod
    def success(data: Optional[Any] = None, msg: str = "操作成功"):
        return {"code": 200, "msg": msg, "data": data}

    @staticmethod
    def fail(msg: str = "操作失败", code: int = 400):
        return {"code": code, "msg": msg, "data": None}

    @staticmethod
    def unauthorized(msg: str = "请先登录"):
        return Res.fail(msg, 401)

    @staticmethod
    def forbidden(msg: str = "权限不足"):
        return Res.fail(msg, 403)
```

  { path: '/admin/daily', name: 'AdminDaily', component: () => import('@/views/admin/Daily.vue'), meta: { admin: true } },
  { path: '/admin/order', name: 'AdminOrder', component: () => import('@/views/admin/Order.vue'), meta: { admin: true } },

  { path: '/admin/config', name: 'AdminConfig', component: () => import('@/views/admin/Config.vue'), meta: { admin: true } },

const router = createRouter({
  history: createWebHistory(),
  routes
})

// 简单权限守卫（可后续增强）
router.beforeEach((to, from, next) => {
  const token = localStorage.getItem('token')
  if (to.meta.login && !token) return next('/login')
  next()
})

export default router
